import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
	<Card title="On screen before the decode finishes" icon="rocket">
		The embedded thumbnail is drawn in single-digit milliseconds while the
		full image decodes behind it, and the neighbours are prefetched. A
		24-megapixel JPEG reaches the screen in about 128 ms from process start -
		a figure a test holds to, so it cannot quietly slip.
	</Card>
	<Card title="The colours the file actually means" icon="approve-check">
		nitid reads the profile out of the file, asks Windows what the display
		is, and converts between them in the shader. A file with no profile is
		passed through untouched, exactly as every browser shows it.
	</Card>
	<Card title="HDR that follows the display" icon="star">
		On an HDR display the output is extended-range linear light, so
		highlights drive the headroom instead of clipping at white. Turn HDR on
		or off in Windows and the surface is reconfigured without a restart.
	</Card>
	<Card title="Every modern still format" icon="document">
		JPEG, PNG, WebP, JPEG XL, HEIC, AVIF, SVG, GIF, BMP and TIFF - all
		decoded in pure Rust, and the animated ones play. The format is decided
		by the bytes rather than the extension, and the heavy decoders run in a
		separate process with no network in either direction.
	</Card>
</CardGrid>
